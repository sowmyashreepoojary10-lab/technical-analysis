#Step 4 — Rank and split into long/short candidates
n_holdings = 10

def get_long_short(momentum_row, n=n_holdings):
    row = momentum_row.dropna()
    if len(row) < 2 * n:
        n = len(row) // 2
    ranked = row.sort_values(ascending=False)
    return ranked.head(n).index.tolist(), ranked.tail(n).index.tolist()

longs, shorts = get_long_short(momentum.iloc[-1])
print("Longs:", longs)
print("Shorts:", shorts)

#Step 5 — Backtest the long-short strategy (full history, not just latest month)
monthly_returns = monthly_prices.pct_change()
strategy_returns, dates = [], []

for i in range(len(momentum) - 1):
    date, next_date = momentum.index[i], momentum.index[i+1]
    if next_date not in monthly_returns.index:
        continue
    longs, shorts = get_long_short(momentum.loc[date])
    if not longs or not shorts:
        continue
    long_ret = monthly_returns.loc[next_date, longs].mean()
    short_ret = monthly_returns.loc[next_date, shorts].mean()
    strategy_returns.append(long_ret - short_ret)
    dates.append(next_date)

strategy_returns = pd.Series(strategy_returns, index=dates, name="Momentum L/S")

#Step 6 — Metrics: long-short vs benchmark

def annualize_return(r): return (1 + r.mean())**12 - 1
def annualize_vol(r): return r.std() * np.sqrt(12)
def sharpe_ratio(r, rf=0.06): return ((r - rf/12).mean() / r.std()) * np.sqrt(12)
def max_drawdown(cum): 
    running_max = cum.cummax()
    return ((cum - running_max) / running_max).min()

cumulative_strategy = (1 + strategy_returns).cumprod()
print(f"Annualized Return: {annualize_return(strategy_returns):.2%}")
print(f"Sharpe Ratio: {sharpe_ratio(strategy_returns):.2f}")
print(f"Max Drawdown: {max_drawdown(cumulative_strategy):.2%}")

#Step 9 — Technical indicators (Stage 4 of the funnel)
def calculate_rsi(prices, period=14):
    delta = prices.diff()
    gain = delta.where(delta > 0, 0)
    loss = -delta.where(delta < 0, 0)
    avg_gain = gain.ewm(alpha=1/period, min_periods=period, adjust=False).mean()
    avg_loss = loss.ewm(alpha=1/period, min_periods=period, adjust=False).mean()
    rs = avg_gain / avg_loss
    return 100 - (100 / (1 + rs))

def calculate_roc(prices, period=12):
    return prices.pct_change(period) * 100

def calculate_macd(prices, fast=12, slow=26, signal=9):
    ema_fast = prices.ewm(span=fast, adjust=False).mean()
    ema_slow = prices.ewm(span=slow, adjust=False).mean()
    macd_line = ema_fast - ema_slow
    signal_line = macd_line.ewm(span=signal, adjust=False).mean()
    return macd_line, signal_line, macd_line - signal_line

def calculate_adx(high, low, close, period=14):
    plus_dm = high.diff(); minus_dm = -low.diff()
    plus_dm[plus_dm < 0] = 0; minus_dm[minus_dm < 0] = 0
    plus_dm[(plus_dm - minus_dm) < 0] = 0; minus_dm[(minus_dm - plus_dm) < 0] = 0
    tr = pd.concat([high-low, (high-close.shift()).abs(), (low-close.shift()).abs()], axis=1).max(axis=1)
    atr = tr.ewm(alpha=1/period, min_periods=period, adjust=False).mean()
    plus_di = 100 * (plus_dm.ewm(alpha=1/period, min_periods=period, adjust=False).mean() / atr)
    minus_di = 100 * (minus_dm.ewm(alpha=1/period, min_periods=period, adjust=False).mean() / atr)
    dx = 100 * (plus_di - minus_di).abs() / (plus_di + minus_di)
    return dx.ewm(alpha=1/period, min_periods=period, adjust=False).mean()

def calculate_52w_high_proximity(prices, window=252):
    return (prices / prices.rolling(window).max()) * 100

# 14 — Define the resistance/breakout function
def calculate_resistance_breakout(close, window=60):
    resistance = close.shift(1).rolling(window).max()
    breakout = close > resistance
    return resistance, breakout
